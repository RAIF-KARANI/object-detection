# VisionTrack dependencies (tested on Windows 11, Python 3.12.9, RTX 4060 Laptop GPU)
#
# The extra index is PyTorch's own package server for CUDA 13.2 builds.
# Without it, pip on Windows installs the CPU-only torch and the GPU is never used.
--extra-index-url https://download.pytorch.org/whl/cu132

torch==2.15.0+cu132          # deep learning engine; runs YOLO on the GPU
torchvision==0.29.2+cu132    # image ops used internally by ultralytics (must match torch)
ultralytics==8.4.215         # YOLO26n detector + built-in ByteTrack tracker
opencv-python==5.1.0.95      # video/webcam input, drawing, display window
numpy==2.5.4                 # array maths (each video frame is a NumPy array)
pandas==3.1.1                # saving results to CSV
